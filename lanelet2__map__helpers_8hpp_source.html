85 template <typename Iter>
86 static Ogre::Vector3 getNormal(Iter it, Iter begin, Iter end) {
87 Ogre::Vector3 zero(0, 0, 0);
88 auto dir_before = zero;
89 int i = 1;
90 while (it != begin && dir_before == zero) {
91 dir_before = *it - *std::prev(it, i);
92 i++;
93 }
94
95 auto dir_after = zero;
96 i = 1;
97 while (std::next(it) != end && dir_after == zero) {
98 dir_after = *std::next(it, i) - *it;
99 i++;
100 }
101
102 if (dir_before == zero && dir_after == zero) {
103 return zero;
104 }
105 if (dir_after == zero) {
106 dir_after = dir_before;
107 }
108 if (dir_before == zero) {
109 dir_before = dir_after;
110 }
111
112 auto dir_combined = dir_after * (1 / dir_after.length()) + dir_before * (1 / dir_before.length());
113 dir_combined *= 1 / dir_combined.length();
114 return Ogre::Vector3(dir_combined.y, -dir_combined.x, 0);
115 }
