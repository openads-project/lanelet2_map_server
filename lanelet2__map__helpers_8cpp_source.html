23void Lanelet2MapHelpers::clearObjects() {
24 for (auto object : owner_.objects_) {
25 switch (object.first) {
26 case ObjectClassification::MAP:
27 case ObjectClassification::AREA:
28 case ObjectClassification::PARKINGAREA:
29 case ObjectClassification::SEPERATOR:
30 case ObjectClassification::STOPLINE:
31 case ObjectClassification::TRAFFICLIGHT:
32 case ObjectClassification::LANEFILL:
33 case ObjectClassification::SIDEWALK:
34 case ObjectClassification::CROSSWALK: {
35 auto man_object = dynamic_cast<Ogre::ManualObject*>(object.second);
36 if (man_object) {
37 man_object->detachFromParent();
38 owner_.scene_manager_->destroyManualObject(man_object);
39 }
40 break;
41 }
42
43 case ObjectClassification::LANELETID: {
44 auto mov_text = dynamic_cast<rviz_rendering::MovableText*>(object.second);
45 if (mov_text) {
46 mov_text->detachFromParent();
47 // MovableText is created directly and attached to the scene node, so it is destroyed directly here.
48 // NOLINTNEXTLINE(cppcoreguidelines-owning-memory)
49 delete mov_text;
50 }
51 break;
52 }
53
54 default: {
55 if (object.second) {
56 object.second->detachFromParent();
57 // Fallback for directly owned OGRE objects not handled by the scene manager helpers above.
58 // NOLINTNEXTLINE(cppcoreguidelines-owning-memory)
59 delete object.second;
60 }
61 }
62 }
63 }
64
65 Lanelet2Map::manual_object_counter_ = 0;
66 owner_.objects_.clear();
67}

129void Lanelet2MapHelpers::addRegulatoryElements(const lanelet::ConstLanelet& lanelet, Ogre::SceneNode* parent_node) {
130 Ogre::SceneNode* regulatory_elements_node = parent_node->createChildSceneNode();
131 auto regulatory_elements = lanelet.regulatoryElements();
132 auto traffic_light_regelems = lanelet.regulatoryElementsAs<lanelet::TrafficLight>();
133
134 if (owner_.rend_opts_.renderStopLines) {
135 for (auto&& reg_element : regulatory_elements) {
136 auto ref_lines = reg_element.get()->getParameters<lanelet::ConstLineString3d>(lanelet::RoleName::RefLine);
137 attachRefLinesToSceneNode(ref_lines, regulatory_elements_node);
138 }
139 }
140
141 if (owner_.rend_opts_.renderTrafficLights) {
142 for (auto&& traffic_light : traffic_light_regelems) {
143 auto traffic_lights = traffic_light.get()->getParameters<lanelet::ConstPolygon3d>(lanelet::RoleName::Refers);
144 attachTrafficLightsToSceneNode(traffic_lights, regulatory_elements_node);
145 }
146 }

149void Lanelet2MapHelpers::attachRefLinesToSceneNode(std::vector<lanelet::ConstLineString3d>& stop_lines,
150 Ogre::SceneNode* parent_node) {
151 Ogre::ManualObject* stop_lines_manual_object =
152 owner_.scene_manager_->createManualObject("llet_object_" + std::to_string(Lanelet2Map::manual_object_counter_++));
153 stop_lines_manual_object->begin(owner_.material_line_->getName(), Ogre::RenderOperation::OT_TRIANGLE_LIST, "rviz_rendering");
154 stop_lines_manual_object->setRenderQueueGroup(Ogre::RENDER_QUEUE_MAIN + 6);
155 for (auto&& stop_line : stop_lines) {
156 auto line = ogreLineFromLLetLineString(stop_line);
157 drawLine(line, stop_lines_manual_object, owner_.rend_opts_.colorStopLine, owner_.rend_opts_.stopLineWidth,
158 owner_.rend_opts_.zStopLine);
159 }
160 // OGRE 1.x exposes section counts through this deprecated API; RViz still uses this vendor version.
161 // NOLINTNEXTLINE(clang-diagnostic-deprecated-declarations)
162 if (stop_lines_manual_object->getNumSections()) {
163 parent_node->attachObject(stop_lines_manual_object);
164 owner_.objects_.push_back(std::make_pair(ObjectClassification::STOPLINE, stop_lines_manual_object));
165 }
166 stop_lines_manual_object->end();
167}

189void Lanelet2MapHelpers::attachLaneletIdToSceneNode(const lanelet::ConstLanelet& lanelet, Ogre::SceneNode* parent_node) {
190 Ogre::SceneNode* child_node = parent_node->createChildSceneNode();
191
192 auto* msg = new rviz_rendering::MovableText(std::to_string(lanelet.id()));
193 msg->setCharacterHeight(boost::numeric_cast<Ogre::Real>(owner_.rend_opts_.characterHeight));
194 msg->setColor(owner_.rend_opts_.colorLaneletId);
195 msg->setTextAlignment(rviz_rendering::MovableText::H_CENTER, rviz_rendering::MovableText::V_ABOVE);
196
197 lanelet::ConstPoint3d text_pos(
198 lanelet::utils::getId(), lanelet.centerline()[lanelet.centerline().size() / 2].x(),
199 lanelet.centerline()[lanelet.centerline().size() / 2].y(),
200 owner_.rend_opts_.threeD ? lanelet.centerline()[lanelet.centerline().size() / 2].z() - 0.2 : -0.2);
201
202 child_node->setPosition(ogreVec3FromLLetPoint(text_pos));
203 child_node->attachObject(msg);
204 owner_.objects_.push_back(std::make_pair(ObjectClassification::LANELETID, msg));
205}

223std::vector<Ogre::Vector3> Lanelet2MapHelpers::ogreLineFromLLetTrafficLight(const lanelet::ConstPolygon3d& polygon) const {
224 if (polygon.empty()) {
225 return {};
226 }
227
228 std::vector<Ogre::Vector3> line;
229 const lanelet::ConstPoint3d lowest_point =
230 *std::min_element(polygon.begin(), polygon.end(),
231 [](const lanelet::ConstPoint3d& p1, const lanelet::ConstPoint3d& p2) { return p1.z() < p2.z(); });
232
233 for (lanelet::ConstPoint3d point : polygon) {
234 line.push_back(ogreVec3FromLLetTrafficLight(point, owner_.rend_opts_.trafficLightHeightAboveGround - lowest_point.z()));
235 }
236 return line;
237}

259std::vector<Ogre::Vector3> Lanelet2MapHelpers::bufferSegment(const std::vector<Ogre::Vector3>& line, double buffer_length) {
260 std::vector<Ogre::Vector3> buffered;
261 buffered.reserve(line.size());
262 assert(line.size() >= 2);
263 for (auto it = line.begin(); it != line.end(); ++it) {
264 auto normal = getNormal(it, line.begin(), line.end());
265 buffered.push_back(*it + normal * boost::numeric_cast<Ogre::Real>(buffer_length / 2.0));
266 }
267 for (auto it = line.rbegin(); it != line.rend(); ++it) {
268 auto normal = getNormal(it, line.rbegin(), line.rend());
269 buffered.push_back(*it + normal * boost::numeric_cast<Ogre::Real>(buffer_length / 2.0));
270 }
271 return buffered;
272}

274void Lanelet2MapHelpers::drawLine(
275 const std::vector<Ogre::Vector3>& line, Ogre::ManualObject* obj, Ogre::ColourValue color, double width, double z_offset) {
276 if (width <= 0 || line.size() < 2) {
277 return;
278 }
279 auto buffered = bufferSegment(line, width);
280 if (z_offset != 0.0) {
281 for (auto& v : buffered) {
282 v.z += boost::numeric_cast<Ogre::Real>(z_offset);
283 }
284 }
285 drawMonoPolygon(buffered, obj, color);
286}

288void Lanelet2MapHelpers::drawArea(const std::vector<Ogre::Vector3>& line,
289 Ogre::ManualObject* obj,
290 Ogre::ColourValue color,
291 double z_offset) {
292 if (line.size() < 2) {
293 return;
294 }
295 if (z_offset != 0.0) {
296 std::vector<Ogre::Vector3> elevated;
297 elevated.reserve(line.size());
298 for (auto v : line) {
299 v.z += boost::numeric_cast<Ogre::Real>(z_offset);
300 elevated.push_back(v);
301 }
302 drawMonoPolygon(elevated, obj, color);
303 return;
304 }
305 drawMonoPolygon(line, obj, color);

308void Lanelet2MapHelpers::drawMonoPolygon(const std::vector<Ogre::Vector3>& poly,
309 Ogre::ManualObject* obj,
310 Ogre::ColourValue color) {
311 if (poly.size() < 3) {
312 return;
313 }
314
315 auto it_left = poly.begin();
316 auto it_right = --poly.end();
317 const auto start_index = obj->getCurrentVertexCount();
318 auto count = 0u;
319 for (; it_right >= it_left; ++it_left, --it_right) {
320 obj->position(*it_left);
321 obj->normal(0, 0, 1);
322 obj->colour(color);
323 if (count >= 2) {
324 assert(obj->getCurrentVertexCount() > start_index + count);
325 obj->triangle(start_index + count - 2, start_index + count - 1, start_index + count);
326 }
327 count++;
328 obj->position(*it_right);
329 obj->normal(0, 0, 1);
330 obj->colour(color);
331 if (count >= 2) {
332 assert(obj->getCurrentVertexCount() > start_index + count);
333 obj->triangle(start_index + count, start_index + count - 1, start_index + count - 2);
334 }
335 count++;
336 }

339void Lanelet2MapHelpers::drawLaneFillStrip(const std::vector<Ogre::Vector3>& left,
340 const std::vector<Ogre::Vector3>& right,
341 Ogre::ManualObject* obj,
342 Ogre::ColourValue color,
343 double z_offset) {
344 if (left.size() < 2 || right.size() < 2) {
345 return;
346 }
347
348 const size_t n_left = left.size();
349 const size_t n_right = right.size();
350 const size_t max_steps = std::max(n_left, n_right) - 1;
351 for (size_t i = 0; i < max_steps; ++i) {
352 double t0 = static_cast<double>(i) / static_cast<double>(max_steps);
353 double t1 = static_cast<double>(i + 1) / static_cast<double>(max_steps);
354
355 size_t i_left0 = static_cast<size_t>(t0 * static_cast<double>(n_left - 1));
356 size_t i_left1 = std::min(static_cast<size_t>(t1 * static_cast<double>(n_left - 1)), n_left - 1);
357 size_t i_right0 = static_cast<size_t>(t0 * static_cast<double>(n_right - 1));
358 size_t i_right1 = std::min(static_cast<size_t>(t1 * static_cast<double>(n_right - 1)), n_right - 1);
359
360 Ogre::Vector3 left0 = left[i_left0];
361 Ogre::Vector3 left1 = left[i_left1];
362 Ogre::Vector3 right0 = right[i_right0];
363 Ogre::Vector3 right1 = right[i_right1];
364 const auto z_offset_real = boost::numeric_cast<Ogre::Real>(z_offset);
365 left0.z += z_offset_real;
366 left1.z += z_offset_real;
367 right0.z += z_offset_real;
368 right1.z += z_offset_real;
369
370 const auto base = obj->getCurrentVertexCount();
371 obj->position(left0);
372 obj->normal(0, 0, 1);
373 obj->colour(color);
374 obj->position(right0);
375 obj->normal(0, 0, 1);
376 obj->colour(color);
377 obj->position(left1);
378 obj->normal(0, 0, 1);
379 obj->colour(color);
380 obj->triangle(base + 0, base + 1, base + 2);
381
382 obj->position(left1);
383 obj->normal(0, 0, 1);
384 obj->colour(color);
385 obj->position(right0);
386 obj->normal(0, 0, 1);
387 obj->colour(color);
388 obj->position(right1);
389 obj->normal(0, 0, 1);
390 obj->colour(color);
391 obj->triangle(base + 3, base + 4, base + 5);
392 }
