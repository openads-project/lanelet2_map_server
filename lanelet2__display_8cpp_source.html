147void Lanelet2Display::initializeMapInterface(rclcpp::Node& parent_node) {
148 std::string name = ll2_server_name_property_->getStdString();
149 if (!name.empty()) {
150 if (name[0] != '/') {
151 name = '/' + name;
152 }
153 lanelet2_map_interface_ = std::make_unique<Lanelet2MapInterface>(parent_node, name);
154 }
155}

160 initializeMapInterface(*rviz_node_);

163void Lanelet2Display::update(float dt, float ros_dt) {
164 if (lanelet2_map_interface_->map_loaded_) {
165 if (!viz_init_) {
166 viz_init_ = visualizeMap();
167 lanelet2_map_interface_->update_pending_ = false;
168 }
169
170 if (lanelet2_map_interface_->update_pending_) {
171 updateVisualization();
172 }
173
174 if (viz_init_) {
175 Q_UNUSED(dt);
176 Q_UNUSED(ros_dt);
177
178 Ogre::Vector3 position;
179 Ogre::Quaternion orientation;
180 if (context_->getFrameManager()->getTransform(lanelet2_map_interface_->map_frame_id_, position, orientation)) {
181 scene_node_->setPosition(position);
182 scene_node_->setOrientation(orientation);
183 setTransformOk();
184 } else {
185 setMissingTransformToFixedFrame(lanelet2_map_interface_->map_frame_id_);
186 map_->getSceneNode()->setVisible(false);
187 }
188 }
189 }
190}

300 QColor color_right = ll_right_col_property_->getColor();
301 color_left.setAlphaF(alpha_property_->getFloat());
302 color_right.setAlphaF(alpha_property_->getFloat());

304 rendering_options_.colorRight = qtToOgre(color_right);

309 rendering_options_.colorSeperator = qtToOgre(color_sep);

313 color_tl.setAlphaF(alpha_property_->getFloat());
